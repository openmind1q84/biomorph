1세대 원시 3마디 유영 생물체에서 시작하여 다채로운 형태와 기관을 가진 생물체로 진화합니다.

진화 제어판

낮음 (안정적) 높음 (역동적)

형태 유전자 구조 (Genes)

현재 중앙 부모 개체의 9가지 유전자 값입니다. 각 유전자는 몸통, 다리/날개, 더듬이, 꼬리 등 구체적인 기관을 조절합니다.

원시 생물체에서 다양한 종으로의 진화 안내

🦐 1세대 (원시 3마디 유영 생물체): 머리(눈·더듬이), 3개의 몸통 마디, 좌우 1쌍의 작은 지느러미, 꼬리를 가진 공통 조상에서 시작합니다.

🧬 9가지 신체 기관 유전자: 돌연변이가 누적되면 몸통 마디 수, 다리/날개 쌍의 개수와 길이, 관절 꺾임, 등 가시/갑각, 꼬리 형태, 보호색 패턴 등이 뚜렷하게 분화합니다.

🌱 자연선택 실험: 특정 환경(심해, 급류, 산호초 등)에 유리한 형질을 가진 자손을 선택해 세대를 거듭하며 적응과 진화를 관찰해 보세요.

// --- EVOLUTIONARY ENGINE: PROTO-ARTHROPOD TO DIVERSE ORGANISMS --- let currentGeneration = 1; // 1세대 초기 유전자: [0,0,0,0,0,0,0,0,0] -> 원시 3마디 유영 생물체 let parentGenome = [0, 0, 0, 0, 0, 0, 0, 0, 0]; let mutationRate = 1; let autoEvolveInterval = null; let ancestorHistory = []; const geneLabels = [ "몸통 마디 수 & 길이", "다리/지느러미 쌍 개수", "다리/날개 길이 & 각도", "머리·눈·더듬이 발달", "등껍질·가시 돌기", "체색 및 보호색 패턴", "꼬리·지느러미 분화", "체형 비율 (유선/넙적)", "외골격 두께 & 발광" ]; function cloneGenome(g) { return [...g]; } function mutateGenome(g) { let mutated = cloneGenome(g); let numMutations = Math.floor(Math.random() * mutationRate) + 1; for (let i = 0; i < numMutations; i++) { let index = Math.floor(Math.random() * mutated.length); let delta = (Math.random() < 0.5 ? 1 : -1) * (Math.random() < 0.8 ? 1 : 2); mutated[index] += delta; if (mutated[index] > 15) mutated[index] = 15; if (mutated[index] < -15) mutated[index] = -15; } return mutated; } // Draw Proto-Arthropod & its evolved morphological variants function drawOrganism(canvas, genome, generation) { const ctx = canvas.getContext('2d'); const width = canvas.width; const height = canvas.height; ctx.clearRect(0, 0, width, height); ctx.save(); ctx.translate(width / 2, height / 2); // Extract morphological traits from the 9 genes const gSegments = genome[0]; // Body segment count & length const gLimbPairs = genome[1]; // Number of limb/fin pairs const gLimbShape = genome[2]; // Limb length, joint angle, wing spread const gHead = genome[3]; // Head size, eyes, antennae/mandibles const gSpines = genome[4]; // Dorsal spines / carapace ridges const gColor = genome[5]; // Hue & pattern type const gTail = genome[6]; // Tail fin / stinger / forked tail const gWidth = genome[7]; // Body width (slender vs broad ray/beetle) const gArmor = genome[8]; // Exoskeleton stroke thickness & bioluminescence // 1. Color Palette Calculation const baseHue = (195 + gColor * 24 + 3600) % 360; const secondaryHue = (baseHue + 40 + (gColor % 3) * 30 + 360) % 360; const bodyFill = `hsl(${baseHue}, 75%, 45%)`; const bodyDark = `hsl(${baseHue}, 80%, 28%)`; const limbColor = `hsl(${secondaryHue}, 82%, 58%)`; const glowColor = `hsl(${secondaryHue}, 95%, 72%)`; // 2. Body Dimensions & Segment Count const numSegments = Math.max(2, Math.min(8, 3 + Math.round(gSegments * 0.4))); const segLength = Math.max(10, Math.min(20, 15 + gSegments * 0.4)); const baseBodyWidth = Math.max(10, Math.min(34, 16 + gWidth * 1.6)); const strokeW = Math.max(1.2, Math.min(4.5, 2 + Math.abs(gArmor) * 0.25)); // Center the creature vertically const totalBodyLen = numSegments * segLength; const startY = -totalBodyLen * 0.35; // Optional Bioluminescent Aura when gArmor > 2 if (gArmor > 2) { ctx.shadowColor = glowColor; ctx.shadowBlur = Math.min(18, gArmor * 2.5); } // 3. Tail Structure (Drawn behind the last segment) const tailY = startY + totalBodyLen; const tailW = Math.max(6, baseBodyWidth * 0.4); ctx.save(); ctx.translate(0, tailY); ctx.fillStyle = limbColor; ctx.strokeStyle = glowColor; ctx.lineWidth = strokeW * 0.8; if (gTail <= -2) { // Scorpion-like segmented stinger tail ctx.beginPath(); ctx.moveTo(0, 0); ctx.quadraticCurveTo(14 + Math.abs(gTail) * 2, 16, 4, 30 + Math.abs(gTail)); ctx.strokeStyle = limbColor; ctx.lineWidth = 3.5; ctx.stroke(); ctx.beginPath(); ctx.arc(4, 30 + Math.abs(gTail), 3.5, 0, Math.PI * 2); ctx.fillStyle = '#f43f5e'; ctx.fill(); } else if (gTail >= 2) { // Broad fan / peacock / fish tail fin const fanSpread = Math.min(36, 12 + gTail * 3); const fanLen = Math.min(34, 16 + gTail * 2); ctx.beginPath(); ctx.moveTo(0, -2); ctx.lineTo(-fanSpread, fanLen); ctx.quadraticCurveTo(0, fanLen - 6, fanSpread, fanLen); ctx.closePath(); ctx.fillStyle = `hsla(${secondaryHue}, 85%, 55%, 0.75)`; ctx.fill(); ctx.stroke(); } else { // Default 1st-gen simple tapering tail fin ctx.beginPath(); ctx.moveTo(-tailW * 0.6, -2); ctx.quadraticCurveTo(0, 22, 0, 26); ctx.quadraticCurveTo(0, 22, tailW * 0.6, -2); ctx.closePath(); ctx.fillStyle = `hsla(${baseHue}, 85%, 60%, 0.6)`; ctx.fill(); ctx.stroke(); } ctx.restore(); // 4. Limbs / Fins / Wings (Drawn attached to body segments) const limbPairs = Math.max(1, Math.min(numSegments, 1 + Math.abs(Math.round(gLimbPairs * 0.6)))); const limbLen = Math.max(12, Math.min(48, 16 + Math.abs(gLimbShape) * 3.2)); const limbAngle = (gLimbShape * 6) * (Math.PI / 180); const isWingMode = gLimbShape >= 3; const isJointedLeg = gLimbShape <= -2; for (let p = 0; p < limbPairs; p++) { // Distribute pairs across segments const segIdx = Math.min(numSegments - 1, Math.floor((p / limbPairs) * numSegments)); const py = startY + segIdx * segLength + segLength * 0.5; const taper = 1 - (segIdx / (numSegments + 1)) * 0.35; const attachX = baseBodyWidth * taper * 0.85; const pairScale = 1 - (p * 0.08); [-1, 1].forEach(dir => { ctx.save(); ctx.translate(dir * attachX, py); ctx.strokeStyle = limbColor; ctx.fillStyle = `hsla(${secondaryHue}, 80%, 55%, 0.45)`; ctx.lineWidth = strokeW; ctx.lineCap = 'round'; if (isWingMode) { // Broad butterfly/ray-like fin or wing const wLen = limbLen * pairScale * 1.15; const wHeight = segLength * 0.9; ctx.beginPath(); ctx.moveTo(0, -wHeight * 0.3); ctx.bezierCurveTo( dir * wLen * 0.6, -wHeight * 1.1 - gLimbShape * 1.5, dir * wLen * 1.1, wHeight * 0.2, dir * wLen * 0.7, wHeight * 0.9 ); ctx.quadraticCurveTo(dir * wLen * 0.3, wHeight * 0.5, 0, wHeight * 0.3); ctx.closePath(); ctx.fill(); ctx.stroke(); } else if (isJointedLeg) { // Articulated arthropod/spider leg with knee joint const l1 = limbLen * 0.55 * pairScale; const l2 = limbLen * 0.65 * pairScale; const kneeX = dir * l1 * Math.cos(limbAngle * 0.4); const kneeY = -6 + l1 * Math.sin(limbAngle * 0.4); const footX = kneeX + dir * l2 * 0.65; const footY = kneeY + l2 * 0.85; ctx.beginPath(); ctx.moveTo(0, 0); ctx.lineTo(kneeX, kneeY); ctx.lineTo(footX, footY); ctx.stroke(); // Joint dot ctx.fillStyle = glowColor; ctx.beginPath(); ctx.arc(kneeX, kneeY, 2, 0, Math.PI * 2); ctx.fill(); } else { // Default 1st-gen paddle fin / simple leg const endX = dir * limbLen * pairScale * Math.cos(limbAngle); const endY = 6 + limbLen * pairScale * Math.sin(limbAngle); ctx.beginPath(); ctx.moveTo(0, -2); ctx.quadraticCurveTo(endX * 0.6, endY - 4, endX, endY); ctx.quadraticCurveTo(endX * 0.5, endY + 4, 0, 3); ctx.closePath(); ctx.fill(); ctx.stroke(); } ctx.restore(); }); } // 5. Segmented Body (Drawn from back to front for natural overlapping) for (let s = numSegments - 1; s >= 0; s--) { const sy = startY + s * segLength + segLength * 0.5; const taper = 1 - (s / (numSegments + 1)) * 0.42; const rx = baseBodyWidth * taper; const ry = segLength * 0.65; // Dorsal spines / carapace spikes if gSpines != 0 if (Math.abs(gSpines) >= 1) { const spikeLen = Math.min(18, Math.abs(gSpines) * 2.2); ctx.fillStyle = limbColor; [-1, 1].forEach(dir => { ctx.beginPath(); ctx.moveTo(dir * (rx - 2), sy - 2); ctx.lineTo(dir * (rx + spikeLen), sy + (gSpines > 0 ? 4 : -4)); ctx.lineTo(dir * (rx - 2), sy + 3); ctx.closePath(); ctx.fill(); }); } // Main segment oval ctx.beginPath(); ctx.ellipse(0, sy, rx, ry, 0, 0, Math.PI * 2); ctx.fillStyle = (s % 2 === 0) ? bodyFill : bodyDark; ctx.fill(); ctx.lineWidth = strokeW; ctx.strokeStyle = glowColor; ctx.stroke(); // Camouflage spots / stripes pattern when gColor is strong if (Math.abs(gColor) >= 2) { ctx.fillStyle = `hsla(${secondaryHue}, 90%, 70%, 0.55)`; ctx.beginPath(); ctx.arc(0, sy, Math.max(2, rx * 0.22), 0, Math.PI * 2); ctx.fill(); } } // 6. Head, Antennae/Mandibles & Eyes (Front of the organism) const headY = startY - 4; const headRx = baseBodyWidth * (0.95 + Math.max(-0.2, gHead * 0.03)); const headRy = Math.max(11, 13 + Math.abs(gHead) * 0.5); // Antennae or Mandibles const antLen = Math.max(8, 10 + Math.abs(gHead) * 2.4); [-1, 1].forEach(dir => { ctx.beginPath(); ctx.strokeStyle = glowColor; ctx.lineWidth = strokeW * 0.85; ctx.lineCap = 'round'; if (gHead >= 0) { // Long sensory antennae ctx.moveTo(dir * headRx * 0.4, headY - headRy * 0.7); ctx.quadraticCurveTo( dir * (headRx * 0.9 + antLen * 0.4), headY - headRy - antLen * 0.6, dir * (headRx * 0.5 + antLen * 0.7), headY - headRy - antLen ); } else { // Beetle-like pincers / mandibles ctx.moveTo(dir * headRx * 0.45, headY - headRy * 0.6); ctx.quadraticCurveTo( dir * (headRx * 0.85), headY - headRy - antLen * 0.7, dir * (headRx * 0.15), headY - headRy - antLen ); } ctx.stroke(); }); // Head Shield ctx.beginPath(); ctx.ellipse(0, headY, headRx, headRy, 0, 0, Math.PI * 2); ctx.fillStyle = bodyFill; ctx.fill(); ctx.lineWidth = strokeW; ctx.strokeStyle = glowColor; ctx.stroke(); // Eyes (Size scales with gHead) const eyeRadius = Math.max(2.5, Math.min(6, 3.2 + Math.abs(gHead) * 0.25)); const eyeX = headRx * 0.52; const eyeY = headY - headRy * 0.35; [-1, 1].forEach(dir => { ctx.fillStyle = '#ffffff'; ctx.beginPath(); ctx.arc(dir * eyeX, eyeY, eyeRadius, 0, Math.PI * 2); ctx.fill(); ctx.fillStyle = '#0f172a'; ctx.beginPath(); ctx.arc(dir * eyeX, eyeY - eyeRadius * 0.2, eyeRadius * 0.52, 0, Math.PI * 2); ctx.fill(); }); ctx.restore(); } function renderUI() { // 1. Render Gene Inspector List const geneListEl = document.getElementById('gene-list'); geneListEl.innerHTML = ''; parentGenome.forEach((val, idx) => { const div = document.createElement('div'); div.className = 'flex items-center justify-between bg-slate-900/60 p-2 rounded-xl border border-slate-700/50'; div.innerHTML = ` ${idx+1}. ${geneLabels[idx]}
`; geneListEl.appendChild(div); }); // 2. Update generation text document.getElementById('txt-generation').textContent = currentGeneration; // 3. Render 3x3 Breeding Grid const gridEl = document.getElementById('breeding-grid'); gridEl.innerHTML = ''; for (let i = 0; i < 9; i++) { const isParent = (i === 4); let genome = isParent ? parentGenome : mutateGenome(parentGenome); const card = document.createElement('div'); card.className = `relative bg-slate-900/80 rounded-2xl p-3 border ${isParent ? 'border-sky-500 ring-2 ring-sky-500/30' : 'border-slate-700 hover:border-slate-500'} flex flex-col items-center justify-between transition group cursor-pointer shadow-lg`; card.innerHTML = `
${isParent ? '현재 부모 개체' : `자손 #${i < 4 ? i+1 : i}`}
[${genome.join(', ')}] ${isParent ? '기준형태' : '선택하기 ➔'}
`; if (!isParent) { card.addEventListener('click', () => { selectOffspring(genome); }); } gridEl.appendChild(card); setTimeout(() => { const canvas = document.getElementById(`canvas-bio-${i}`); if (canvas) drawOrganism(canvas, genome, currentGeneration); }, 10); } renderLineage(); } function selectOffspring(newGenome) { ancestorHistory.push({ gen: currentGeneration, genome: cloneGenome(parentGenome) }); parentGenome = cloneGenome(newGenome); currentGeneration++; renderUI(); } // Open Ancestor Detail Popup let currentModalAncestor = null; function openAncestorModal(item) { currentModalAncestor = item; document.getElementById('ancestor-modal-gen').textContent = `${item.gen}세대 조상`; document.getElementById('ancestor-modal-code').textContent = `[ ${item.genome.join(', ')} ]`; const modal = document.getElementById('modal-ancestor'); modal.classList.remove('hidden'); setTimeout(() => { const canvas = document.getElementById('canvas-ancestor-detail'); if (canvas) drawOrganism(canvas, item.genome, item.gen); }, 10); } function renderLineage() { const container = document.getElementById('lineage-container'); if (ancestorHistory.length === 0) { container.innerHTML = '1세대 원시 3마디 생물체 조상부터 진화 계통수가 기록됩니다. (클릭 시 상세 정보 확인)'; return; } container.innerHTML = ''; ancestorHistory.forEach((item, idx) => { const node = document.createElement('div'); node.className = 'flex items-center space-x-2 bg-slate-900 hover:bg-slate-800 px-3 py-2 rounded-xl border border-slate-700/60 hover:border-purple-500/60 flex-shrink-0 cursor-pointer transition group'; node.title = '클릭하여 확대 이미지 및 전체 유전자 코드 보기'; node.innerHTML = `
${item.gen}세대 조상
[${item.genome.slice(0,3).join(',')}]...
`; node.addEventListener('click', () => openAncestorModal(item)); container.appendChild(node); setTimeout(() => { const canvas = document.getElementById(`canvas-ancestor-${idx}`); if (canvas) drawOrganism(canvas, item.genome, item.gen); }, 10); }); } // Mutation rate slider const mutationSlider = document.getElementById('input-mutation-rate'); const mutationValSpan = document.getElementById('val-mutation-rate'); mutationSlider.addEventListener('input', (e) => { mutationRate = parseInt(e.target.value); mutationValSpan.textContent = mutationRate; }); // Apply custom genes document.getElementById('btn-apply-genes').addEventListener('click', () => { let newGenome = []; for (let i = 0; i < 9; i++) { const val = parseInt(document.getElementById(`gene-input-${i}`).value); if (isNaN(val)) { showMessage('모든 유전자는 유효한 숫자여야 합니다.', 'error'); return; } newGenome.push(val); } parentGenome = newGenome; renderUI(); showMessage('유전자가 성공적으로 적용되었습니다.', 'success'); }); // Reset to generation 1 basic shape document.getElementById('btn-reset').addEventListener('click', () => { if (autoEvolveInterval) toggleAutoEvolve(); currentGeneration = 1; parentGenome = [0, 0, 0, 0, 0, 0, 0, 0, 0]; ancestorHistory = []; renderUI(); showMessage('1세대 원시 3마디 유영 생물체 상태로 초기화되었습니다.', 'info'); }); // Auto-Evolve Toggle const btnAuto = document.getElementById('btn-auto-evolve'); const iconAuto = document.getElementById('icon-auto'); const textAuto = document.getElementById('text-auto'); const selectSpeed = document.getElementById('select-speed'); function toggleAutoEvolve() { if (autoEvolveInterval) { clearInterval(autoEvolveInterval); autoEvolveInterval = null; iconAuto.className = 'fa-solid fa-play'; textAuto.textContent = '자동 진화 시작'; btnAuto.className = 'flex-1 py-2 px-3 bg-sky-600 hover:bg-sky-500 text-white rounded-xl text-xs font-semibold shadow-md transition flex items-center justify-center space-x-1.5'; } else { iconAuto.className = 'fa-solid fa-pause'; textAuto.textContent = '자동 진화 중지'; btnAuto.className = 'flex-1 py-2 px-3 bg-amber-600 hover:bg-amber-500 text-white rounded-xl text-xs font-semibold shadow-md transition flex items-center justify-center space-x-1.5'; let speed = parseInt(selectSpeed.value); autoEvolveInterval = setInterval(() => { parentGenome = mutateGenome(parentGenome); currentGeneration++; renderUI(); }, speed); } } btnAuto.addEventListener('click', toggleAutoEvolve); // Export and Import Modals document.getElementById('btn-export').addEventListener('click', () => { const codeStr = parentGenome.join(', '); navigator.clipboard.writeText(codeStr).then(() => { showMessage(`유전자 코드 [${codeStr}]가 클립보드에 복사되었습니다!`, 'success'); }).catch(() => { showMessage(`유전자 코드: ${codeStr}`, 'info'); }); }); const importModal = document.getElementById('modal-import'); document.getElementById('btn-import').addEventListener('click', () => { document.getElementById('import-textarea').value = parentGenome.join(', '); importModal.classList.remove('hidden'); }); document.getElementById('import-close').addEventListener('click', () => importModal.classList.add('hidden')); document.getElementById('import-cancel').addEventListener('click', () => importModal.classList.add('hidden')); document.getElementById('import-submit').addEventListener('click', () => { const text = document.getElementById('import-textarea').value; const parts = text.split(/[\s,]+/).filter(Boolean).map(Number); if (parts.length !== 9 || parts.some(isNaN)) { showMessage('9자리의 올바른 숫자 유전자 코드를 입력해주세요.', 'error'); return; } parentGenome = parts; importModal.classList.add('hidden'); renderUI(); showMessage('유전자 코드를 성공적으로 불러왔습니다.', 'success'); }); // Help Modal const helpModal = document.getElementById('modal-help'); document.getElementById('btn-help').addEventListener('click', () => helpModal.classList.remove('hidden')); document.getElementById('modal-close').addEventListener('click', () => helpModal.classList.add('hidden')); document.getElementById('btn-help-ok').addEventListener('click', () => helpModal.classList.add('hidden')); // Ancestor Detail Modal Controls const ancestorModal = document.getElementById('modal-ancestor'); document.getElementById('ancestor-close').addEventListener('click', () => ancestorModal.classList.add('hidden')); ancestorModal.addEventListener('click', (e) => { if (e.target === ancestorModal) ancestorModal.classList.add('hidden'); }); document.getElementById('btn-ancestor-copy').addEventListener('click', () => { if (!currentModalAncestor) return; const codeStr = currentModalAncestor.genome.join(', '); const tempInput = document.createElement('textarea'); tempInput.value = codeStr; document.body.appendChild(tempInput); tempInput.select(); document.execCommand('copy'); document.body.removeChild(tempInput); showMessage(`${currentModalAncestor.gen}세대 유전자 코드 [${codeStr}]가 복사되었습니다!`, 'success'); }); document.getElementById('btn-ancestor-download').addEventListener('click', () => { if (!currentModalAncestor) return; const sourceCanvas = document.getElementById('canvas-ancestor-detail'); // Create temporary canvas with dark background for clean image export const exportCanvas = document.createElement('canvas'); ctx.drawImage(sourceCanvas, 0, 0); const link = document.createElement('a'); link.download = `biomorph_gen${currentModalAncestor.gen}.png`; link.href = exportCanvas.toDataURL('image/png'); link.click(); }); // Initial load window.onload = function() { renderUI(); };
